% 图 2.2 端到端数据流 — sequence-like horizontal lifelines
\documentclass[border=4pt]{standalone}
\usepackage{xeCJK}
\setCJKmainfont{Songti SC}
\setCJKsansfont{Heiti SC}
\usepackage{tikz}
\usetikzlibrary{arrows.meta, positioning, calc, shapes.geometric, fit, backgrounds}
\input{_style.tex}

\begin{document}
\begin{tikzpicture}[font=\sffamily, x=22mm, y=-7mm]

% ── 5 条 lifelines（横向一字排开），从左到右是时序起点 ──
\foreach \i/\name/\fill in {%
  0/{用户}/white,
  1/{玄女}/{fuxiBlue!8},
  2/{Shelf}/white,
  3/{门客 cc}/white,
  4/{EventBus}/white,
  5/{SQLite/IM}/white%
} {
  \node[modBox, anchor=south, fill=\fill, minimum width=18mm] (h\i) at (\i, 0) {\name};
  \draw[fuxiGrayLine!60, thick] (\i, -0.4) -- (\i, -10);
}

% ── 时序消息（自上而下时间增长）──
\draw[flow] (h0) -- ++(0,-1) coordinate (a0) -- node[above, font=\scriptsize] {① 用户输入} (a0 -| h1) coordinate (a1);
\node[numBubble, fill=fuxiBlue!8] at ($(a0)!0.5!(a1)+(0,2mm)$) {1};

\draw[flow] (h1 |- 0,-1.6) -- node[above, font=\scriptsize] {② claim\_idle} (h2 |- 0,-1.6);
\draw[flow, color=fuxiBlue] (h1 |- 0,-2.4) -- node[above, font=\scriptsize] {③ A2A dispatch} (h3 |- 0,-2.4);
\node[numBubble, fill=fuxiBlue!8] at ($(h1 |- 0,-2.4)!0.5!(h3 |- 0,-2.4)+(0,2mm)$) {3};

% 门客内部（子进程：cc CLI fork）
\node[ghostBox, anchor=west, minimum width=14mm, font=\scriptsize\itshape] (cli) at ($(h3)+(0.2,-3.6)$) {fork cc CLI 子进程};

% 门客发事件流回 EventBus
\draw[flowEvent] (h3 |- 0,-4.8) -- node[above, font=\scriptsize, color=fuxiBlue] {④ event publish} (h4 |- 0,-4.8);
\node[numBubble, fill=fuxiBlue!8] at ($(h3 |- 0,-4.8)!0.5!(h4 |- 0,-4.8)+(0,2mm)$) {4};

% EventBus 同时发给 SQLite + IM
\draw[flowEvent, color=fuxiBlue] (h4 |- 0,-5.6) -- node[above, font=\scriptsize] {⑤ WAL append} (h5 |- 0,-5.6);
\draw[flowDup] (h4 |- 0,-6.4) -- node[above, font=\scriptsize] {⑤' broadcast → IM} (h5 |- 0,-6.4);

% 门客返回结果
\draw[flowEvent] (h3 |- 0,-7.4) -- node[above, font=\scriptsize] {⑥ result + done} (h4 |- 0,-7.4);
\draw[flowDup] (h4 |- 0,-8.2) -- (h5 |- 0,-8.2);

% 玄女通过 EventBus 收到通知，更新状态给用户
\draw[flowDup] (h4 |- 0,-9) -| (h1 |- 0,-9.5);
\draw[flow] (h1 |- 0,-9.5) -- node[above, font=\scriptsize] {⑦ stream-back} (h0 |- 0,-9.5);
\node[numBubble, fill=fuxiBlue!8] at ($(h1 |- 0,-9.5)!0.5!(h0 |- 0,-9.5)+(0,2mm)$) {7};

% 时间轴
\draw[->, color=fuxiGrayDark, thick] (-0.7, -0.8) -- (-0.7, -10) node[midway, rotate=90, above, font=\scriptsize\itshape, color=fuxiGrayDark] {时间};

\end{tikzpicture}
\end{document}
