  {cc 门客 \#1\\\scriptsize claude-code};

  {cc 门客 \#2\\\scriptsize claude-code};

  {codex 门客\\\scriptsize OpenAI codex};

\node[highlightBox, below=20mm of shelf, minimum width=38mm, minimum height=12mm] (bus)

  {SQLite WAL\\\scriptsize 单一真相源};

% ── Boundaries ──────────────────────────────────────
\begin{scope}[on background layer]
  \node[externalBoundary, fit=(im) (tui), inner sep=6pt,
    label={[font=\scriptsize\itshape, text=fuxiGrayDark, yshift=1pt]above:{用户终端/浏览器}}] {};
  \node[externalBoundary, fit=(cc1) (cc2) (codex), inner sep=6pt,
    label={[font=\scriptsize\itshape, text=fuxiGrayDark, yshift=1pt]above:{子进程（CLI 头无 GUI）}}] {};

% ── Control flow: horizontal and local vertical only ─
\draw[flow] (im.east) -- node[above=1pt, font=\scriptsize] {REST/WS} (xuannv.west);
\draw[bidir] (xuannv.south) -- (shelf.north);

\coordinate (workerHub) at ($(cc1.west)!0.5!(codex.west)+(-6mm,0)$);
\draw[bidir] (xuannv.east) -- node[above=1pt, font=\scriptsize] {A2A} (workerHub);
\draw[flow] (workerHub) |- (cc1.west);
\draw[flow] (workerHub) -- (cc2.west);
\draw[flow] (workerHub) |- (codex.west);

% ── Event flow: 编排层 publish — shelf 单一通路（玄女经 shelf 间接发布）─
\draw[flowEvent] (shelf.south) -- node[right=1pt, font=\scriptsize, color=fuxiBlue] {publish} (bus.north);

% ── 3 条 worker → bus 错开列与入口 y，避免共线 + 无 U 拐 ─
% cc1（顶）：列 +10mm, 入 bus 上沿 +4mm — 路径最长
\draw[flowEvent] (cc1.east) -- ++(10mm,0) |- ($(bus.east)+(0,4mm)$);
% cc2（中）：列 +7mm, 入 bus 中线
\draw[flowEvent] (cc2.east) -- ++(7mm,0)  |- (bus.east);
% codex（底）：列 +4mm, 入 bus 下沿 -4mm — 路径最短
\draw[flowEvent] (codex.east) -- ++(4mm,0) |- ($(bus.east)+(0,-4mm)$);

% ── Subscription and storage paths ──────────────────
\draw[flowDup] (bus.west) -| ($(im.south west)+(-4mm,-4mm)$) |- (im.west);
\draw[flow] (tui.east) -| ($(bus.west)+(-5mm,0)$) -- (bus.west);
\draw[flowEvent] (bus.south) -- (sqlite.north);

% ── Legend ─────────────────────────────────────────
\node[anchor=north east, font=\scriptsize, align=left, draw=fuxiGrayLine!40, fill=white,
      inner sep=4pt, line width=0.3pt] at ($(sqlite.south east)+(44mm,-1mm)$) {
