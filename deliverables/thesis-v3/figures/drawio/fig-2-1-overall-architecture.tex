% ── 门客（右侧三只）─────────────────────────────────
\node[modBox, right=14mm of xuannv, minimum width=24mm] (cc1) {cc 门客 \#1\\\scriptsize claude-code};
\node[modBox, below=2mm of cc1, minimum width=24mm] (cc2) {cc 门客 \#2\\\scriptsize claude-code};
\node[modBox, below=2mm of cc2, minimum width=24mm] (codex) {codex 门客\\\scriptsize OpenAI codex};

% ── 持久化层（最下）────────────────────────────────
\node[ghostBox, below=10mm of bus, minimum width=28mm] (sqlite) {SQLite WAL\\\scriptsize 单一真相源};

  \node[externalBoundary, fit=(cc1) (cc2) (codex), inner sep=4pt, label={[font=\scriptsize\itshape, text=fuxiGrayDark]above:{子进程（CLI 头无 GUI）}}] (subprocs) {};

% ── 数据流箭头 ──────────────────────────────────────
% 用户 → 玄女
\draw[flow] (im.east) -- node[above, font=\scriptsize, pos=0.5] {REST/WS} (xuannv.west);
\draw[flow] (tui.east) -- (bus.west);

\draw[bidir] (xuannv.east) -- node[above, font=\scriptsize] {A2A} (cc1.west);

\draw[flowDup] (bus.west) -| ($(im.south)+(0mm,-3mm)$) |- (im.east);
